#!/usr/bin/env bash
# Detached session driver: SOP So400m vs Large pair (design docs/sop_so400m_pair_design_2026-10-16.md).
# Smokes (17 updates, both arms) must succeed before the 1000-update pair. No retries.
set -euo pipefail
R=/home/riomus/runs; D=$R/sfora-sop-so400m-pair-src-v3; PY=/home/riomus/group-learning/.venv/bin/python
HUB=/home/riomus/.cache/huggingface/hub
declare -A MODEL=([large]=$HUB/models--google--siglip2-large-patch16-256/snapshots/787800c8990e6f058423089178e718139608408c
                  [so400_280]=$HUB/models--google--siglip2-so400m-patch14-384/snapshots/e8e487298228002f3d8a82e0cd5c8ea9c567f57f)
declare -A FEAT=([large]=$R/sfora-siglip2-train-28e19203 [so400_280]=$R/sfora-sop-so400m-train-features-v1)
TILE=/home/riomus/toolchains/cuda-13.4-wheel-env/lib/python3.12/site-packages/nvidia/cu13/bin/tileiras
say() { echo "SUPERCLASS $(date -u +%FT%TZ) $*"; }
say "BEGIN pid=$$ cgroup=$(cat /proc/$$/cgroup) commit=$(cat $D/COMMIT)"
run() {  # substrate updates
  local s=$1 u=$2 name=sfora-sop-res280sc-$1-179024-$2-v1 extra=()
  [[ $u == 1000 ]] && extra=(--evaluate)
  test ! -e $R/$name
  say "START $s $u"
  systemd-run --user --wait --collect --unit=$name -p StandardOutput=append:$R/$name.log -p StandardError=append:$R/$name.log \
    -p WorkingDirectory=$D/scripts -p MemoryMax=24G -p MemorySwapMax=0 -p RuntimeMaxSec=7200 \
    --setenv=PYTHONPATH=$D/src:$D/scripts --setenv=HF_HUB_OFFLINE=1 --setenv=CUTILE_TILEIRAS_PATH=$TILE \
    --setenv=PATH=$(dirname $TILE):/usr/local/bin:/usr/bin:/bin --setenv=CUDA_VISIBLE_DEVICES=0 \
    /usr/bin/flock -n $R/.sfora-siglip2-gpu.lock /usr/bin/flock -n /home/riomus/.sfora-siglip2-gpu.lock \
    /usr/bin/time -v $PY $D/scripts/train_sop_so400m_res280_superclass_pilot.py --substrate $s \
    --model-snapshot ${MODEL[$s]} --candidate-dir ${FEAT[$s]} \
    --unicom-l14-archive /home/riomus/sfora-relational-sop-e1/unicom-l14-sop-v1.npz \
    --dataset-root /home/riomus/datasets/Stanford_Online_Products \
    --native-library /home/riomus/sfora-rc5-pointer-b7c57022/libsfora_cutile_int8_score_sha39602d0e.so \
    --output-dir $R/$name --arm float_rank --seed 179024 --updates $u --batch-size 64 --workers 2 \
    --rank-coefficient 8.0 --train-vision-dtype bf16 --coverage-first-schedule --freeze-lower-stack \
    --member-bank --member-bank-preflight $R/sfora-siglip2-arcface-1000-v10-55040a7/member_bank_preflight_v1.json \
    --expected-member-bank-preflight-sha256 54e806715e76b2caa7e877d55b0fecbdc921718386bce845e04367164828624c \
    --member-bank-cost-receipt $R/sfora-siglip2-member-bank-cost-v1/receipt_v2.json \
    --expected-member-bank-cost-sha256 8f6867ff4de768ffd3bfa108cb86d7537b913d6ae5cb4f8cb16a43bc87f741a9 \
    "${extra[@]}" || { say "HOLD $s $u failed"; exit 1; }
  say "DONE $s $u receipt=$(sha256sum $R/$name/receipt.json | cut -c1-64)"
}

run so400_280 1000
say "ALLDONE"
